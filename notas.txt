<?php
// Ruta al archivo Excel generado
$excelPath = __DIR__ . '/reporte.xlsx';

$libreOfficePath = '"C:\Program Files\LibreOffice\program\soffice.exe"';

// Comando para convertir
$command = $libreOfficePath . ' --headless --convert-to pdf --outdir ' . escapeshellarg($pdfOutputDir) . ' ' . escapeshellarg($excelPath);

exec($command, $output, $resultCode);

    echo "✅ PDF generado correctamente.";
